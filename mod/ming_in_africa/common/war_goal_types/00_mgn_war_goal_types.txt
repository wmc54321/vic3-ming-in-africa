mgn_chinese_subjugation = {
	icon = "gfx/interface/icons/war_goals/make_tributary.dds"

	kind = custom

	settings = {
		require_target_be_part_of_war
		turns_into_subject
		requires_interest

		conflicts_with_make_subject
		validate_conflicts_make_subject
	}

	execution_priority = 12

	contestion_type = control_target_country_capital

	target_type = country

	possible = {
		ROOT = { mgn_is_chinese_country = yes }
		scope:target_country ?= {
			mgn_is_chinese_country = yes
			NOT = { this = root }
		}
	}

	valid = {
		ROOT = { mgn_is_chinese_country = yes }
		scope:target_country ?= {
			mgn_is_chinese_country = yes
			NOT = { this = root }
		}
	}

	maneuvers = {
		value = 0
		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 10
		}
	}

	infamy = {
		value = 0
		add = {
			desc = "INFAMY_BASE_VALUE"
			value = 8
		}
	}

	on_enforced = {
		if = {
			limit = {
				scope:target_country ?= { is_country_type = recognized }
			}
			ROOT = {
				create_diplomatic_pact = {
					country = scope:target_country
					type = dominion
				}
			}
		}
		else_if = {
			limit = {
				scope:target_country ?= { is_country_type = unrecognized }
			}
			ROOT = {
				create_diplomatic_pact = {
					country = scope:target_country
					type = tributary
				}
			}
		}
		else = {
			ROOT = {
				create_diplomatic_pact = {
					country = scope:target_country
					type = vassal
				}
			}
		}
		mgn_on_two_chinas_subject_relation = { OVERLORD = ROOT SUBJECT = scope:target_country }
	}

	ai = {
		is_significant_demand = yes
	}
}

mgn_two_chinas_independence = {
	icon = "gfx/interface/icons/war_goals/independence.dds"

	kind = independence

	settings = {
		require_target_be_part_of_war
		can_add_for_other_country
		preserve_on_switching_sides

		validate_conflicts_war_goals_all
		validate_independence
	}

	execution_priority = 100

	contestion_type = control_all_own_states

	target_type = country

	valid = {
		root.overlord = scope:target_country
		OR = {
			AND = { ROOT = { c:MGN ?= THIS } scope:target_country = { c:CHI ?= THIS } }
			AND = { ROOT = { c:CHI ?= THIS } scope:target_country = { c:MGN ?= THIS } }
		}
	}

	maneuvers = {
		value = 0
		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 20
		}
	}

	infamy = {
		value = 0
	}

	on_enforced = {
		remove_variable = mgn_bitter_peace_started
		mgn_restart_two_chinas_journal = yes
		scope:target_country ?= { mgn_restart_two_chinas_journal = yes }
	}

	ai = {
		is_significant_demand = yes
	}
}

mgn_open_chinese_migration = {
	icon = "gfx/interface/icons/war_goals/regime_change.dds"

	settings = {
		require_target_be_part_of_war
		can_add_for_other_country
		requires_interest
	}

	execution_priority = 19

	contestion_type = control_target_country_capital

	target_type = country

	possible = {
		ROOT = {
