law_mgn_revolutionary_committees = {
	group = lawgroup_distribution_of_power
	parent = law_single_party_state

	icon = "gfx/interface/icons/law_icons/single_party_state.dds"

	progressiveness = 100

	unlocking_technologies = {
		political_agitation
	}

	is_visible = {
		mgn_can_use_revolutionary_committees = yes
	}

	can_enact = {
		mgn_can_use_revolutionary_committees = yes
		has_law_or_variant = law_type:law_council_republic
		OR = {
			custom_tooltip = {
				text = ruler_in_party_tt
				exists = ruler
				ruler = {
					this.interest_group ?= { has_party = yes }
				}
			}
			custom_tooltip = {
				text = sps_no_party_tt
				country_has_voting_franchise = no
			}
		}
	}

	can_impose = {
		always = no
	}

	modifier = {
		country_legitimacy_govt_total_votes_add = 100
		country_legitimacy_govt_size_add = 2
		country_legitimacy_ideological_incoherence_mult = -0.25
		country_authority_add = 250
		country_voting_power_base_add = 50
		interest_group_in_government_attraction_mult = 0.20
		country_electoral_confidence_impact_mult = -1.0
		interest_group_ig_trade_unions_pol_str_mult = 0.15
		interest_group_ig_armed_forces_pol_str_mult = 0.15
	}

	on_enact = {
		custom_tooltip = mgn_revolutionary_committees_special_unlocks_tt
	}

	on_activate = {
		set_only_legal_party_from_ig = ruler.interest_group
		custom_tooltip = {
			text = sps_government_reshuffle_tt
			every_interest_group = {
				limit = {
					is_in_government = yes
					NOT = { has_party = yes }
				}
				remove_ruling_interest_group = yes
			}
			every_interest_group = {
				limit = {
					is_in_government = no
					has_party = yes
				}
				add_ruling_interest_group = yes
			}
		}
		if = {
			limit = {
				game_date > 1836.1.1
				country_has_voting_franchise = yes
			}
			initialize_electoral_confidence = yes
		}
		every_scope_state = {
			mgn_refresh_regime_state_name = yes
		}
	}

	on_deactivate = {
		unset_only_legal_party = yes
		update_party_support = yes
		if = {
			limit = { has_variable = fatherland_front_var }
			remove_variable = fatherland_front_var
		}
		every_scope_state = {
			mgn_refresh_regime_state_name = yes
		}
	}

	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = {
		value = 0
	}
}
