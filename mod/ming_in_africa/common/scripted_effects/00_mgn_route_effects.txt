mgn_refresh_two_chinas_status = {
	if = {
		limit = { mgn_two_chinas_independent_rivals = yes }
		mgn_restart_two_chinas_journal = yes
	}
	else_if = {
		limit = { mgn_is_two_chinas_leader = yes }
		if = {
			limit = { c:MGN ?= THIS }
			mgn_activate_ming_rebuild_state = yes
		}
		else_if = {
			limit = { c:CHI ?= THIS }
			mgn_activate_qing_rebuild_state = yes
		}
	}
	else_if = {
		limit = { mgn_is_two_chinas_subject = yes }
		mgn_activate_bitter_peace = yes
	}
}

mgn_make_brotherly_subject = {
	if = {
		limit = { $OVERLORD$ = { is_country_type = unrecognized } }
		$OVERLORD$ = {
			create_diplomatic_pact = {
				country = $SUBJECT$
				type = tributary
			}
		}
	}
	else_if = {
		limit = { $SUBJECT$ = { is_country_type = unrecognized } }
		$OVERLORD$ = {
			create_diplomatic_pact = {
				country = $SUBJECT$
				type = protectorate
			}
		}
	}
	else = {
		$OVERLORD$ = {
			create_diplomatic_pact = {
				country = $SUBJECT$
				type = protectorate
			}
		}
	}
	mgn_on_two_chinas_subject_relation = { OVERLORD = $OVERLORD$ SUBJECT = $SUBJECT$ }
}

mgn_on_two_chinas_subject_relation = {
	if = {
		limit = {
			OR = {
				AND = {
					$OVERLORD$ = { c:MGN ?= THIS }
					$SUBJECT$ = { c:CHI ?= THIS }
				}
				AND = {
					$OVERLORD$ = { c:CHI ?= THIS }
					$SUBJECT$ = { c:MGN ?= THIS }
				}
			}
		}
		$SUBJECT$ = {
			mgn_activate_bitter_peace = yes
		}
		$OVERLORD$ = {
			if = {
				limit = { c:MGN ?= THIS }
				mgn_activate_ming_rebuild_state = yes
			}
			else_if = {
				limit = { c:CHI ?= THIS }
				mgn_activate_qing_rebuild_state = yes
			}
		}
	}
}
