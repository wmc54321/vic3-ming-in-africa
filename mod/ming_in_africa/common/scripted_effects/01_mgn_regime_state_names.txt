mgn_refresh_regime_state_name = {
	if = {
		limit = { has_game_rule = no_dynamic_naming }
		if = {
			limit = { has_variable = mgn_regime_state_name_applied }
			mgn_restore_regime_state_name = yes
		}
	}
	else_if = {
		limit = {
			exists = owner
			owner = { has_law = law_type:law_mgn_revolutionary_committees }
		}
		mgn_apply_revolutionary_state_name = yes
	}
	else_if = {
		limit = {
			exists = owner
			owner = { has_law = law_type:law_mgn_national_tutelage }
		}
		mgn_apply_national_tutelage_state_name = yes
	}
	else_if = {
		limit = {
			state_region = s:STATE_LOWER_EGYPT
			exists = owner
			owner = {
				OR = {
					c:MGN ?= THIS
					c:CHI ?= THIS
					global_var:chinese_central_government ?= THIS
				}
				capital.state_region = s:STATE_LOWER_EGYPT
				OR = {
					has_law = law_type:law_monarchy
					has_law = law_type:law_social_monarchy
				}
			}
		}
		set_state_name = STATE_LOWER_EGYPT_mgn_imperial_residence
		set_variable = mgn_regime_state_name_applied
	}
	else_if = {
		limit = { has_variable = mgn_regime_state_name_applied }
		mgn_restore_regime_state_name = yes
	}
}
