mgn_is_chinese_country = {
	OR = {
		c:CHI ?= THIS
		c:MGN ?= THIS
		global_var:chinese_central_government ?= THIS
		has_journal_entry = je_reunify_china
		country_has_primary_culture = cu:han
		country_has_primary_culture = cu:manchu
		country_has_primary_culture = cu:african_han
		country_has_primary_culture = cu:western_han
	}
}

mgn_is_chinese_chartered_company = {
	OR = {
		has_variable = mgn_company_cannot_form_china
		c:MCC ?= THIS
		c:MDC ?= THIS
		c:MKC ?= THIS
		c:MLF ?= THIS
		c:MSB ?= THIS
		c:MCA ?= THIS
		c:MJP ?= THIS
		c:MSE ?= THIS
		AND = {
			is_country_type = company
			OR = {
				has_law = law_type:law_mgn_heavenly_subjecthood
				country_has_primary_culture = cu:han
				country_has_primary_culture = cu:manchu
				country_has_primary_culture = cu:african_han
				country_has_primary_culture = cu:western_han
			}
		}
	}
}

mgn_qing_counterpart_exists = {
	exists = c:CHI
	NOT = { has_global_variable = china_shatters }
	c:CHI ?= {
		is_country_alive = yes
		NOT = { has_variable = warlord_state }
	}
	trigger_if = {
		limit = { exists = global_var:chinese_central_government }
		global_var:chinese_central_government ?= c:CHI
	}
}

mgn_two_chinas_not_at_war = {
	mgn_two_chinas_exist = yes
	c:MGN ?= {
		is_at_war = no
		is_active_in_diplomatic_play = no
	}
	c:CHI ?= {
		is_at_war = no
		is_active_in_diplomatic_play = no
	}
	OR = {
		AND = {
			c:MGN ?= THIS
			NOT = { has_war_with = c:CHI }
		}
		AND = {
			c:CHI ?= THIS
			NOT = { has_war_with = c:MGN }
		}
	}
}

	OR = {

		has_law = law_type:law_wealth_voting
		has_law = law_type:law_census_voting

	}

mgn_two_chinas_same_unifiable_government = {
	mgn_two_chinas_exist = yes
	OR = {
		# Monarchies, social monarchies, and corporate states may converge as
		# long as both belong to the electoral or the non-electoral camp.
		AND = {
			mgn_government_is_flexible_union_form = yes
			OR = {
				AND = {
					mgn_has_electoral_distribution_for_union = yes
					OR = {
						AND = { c:MGN ?= THIS c:CHI ?= { mgn_government_is_flexible_union_form = yes mgn_has_electoral_distribution_for_union = yes } }
						AND = { c:CHI ?= THIS c:MGN ?= { mgn_government_is_flexible_union_form = yes mgn_has_electoral_distribution_for_union = yes } }
					}
				}
				AND = {
					NOT = { mgn_has_electoral_distribution_for_union = yes }
					OR = {
						AND = { c:MGN ?= THIS c:CHI ?= { mgn_government_is_flexible_union_form = yes NOT = { mgn_has_electoral_distribution_for_union = yes } } }
						AND = { c:CHI ?= THIS c:MGN ?= { mgn_government_is_flexible_union_form = yes NOT = { mgn_has_electoral_distribution_for_union = yes } } }
					}
				}
			}
		}
		# Every other government must use the exact same governance principle.
		AND = {
			NOT = { mgn_government_is_flexible_union_form = yes }
			OR = {
				AND = {
					c:MGN ?= THIS
					c:CHI ?= { has_law = ROOT.active_law:lawgroup_governance_principles.type }
				}
				AND = {
					c:CHI ?= THIS
					c:MGN ?= { has_law = ROOT.active_law:lawgroup_governance_principles.type }
				}
			}
		}
	}
}
